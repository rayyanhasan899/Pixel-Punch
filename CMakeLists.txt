cmake_minimum_required(VERSION 3.16)
project(pixel_punch LANGUAGES CXX)
find_package(SFML 2.6 COMPONENTS graphics window system REQUIRED)
add_executable(pixel-punch PixelPunch.cpp)
target_compile_features(pixel-punch PRIVATE cxx_std_17)
target_link_libraries(pixel-punch PRIVATE sfml-graphics sfml-window sfml-system)
if(MSVC)
  target_compile_options(pixel-punch PRIVATE /W4 /utf-8)
endif()
